手工安装 ​

解压缩并移动到目标目录

bash
# JDK 11
tar -xzf bellsoft-jdk11.0.32.1+1-linux-amd64.tar.gz
sudo mv jdk-11.0.32.1 jdk-11
sudo mv jdk-11 /data/tools/jdk
# JDK 21
tar -xzf bellsoft-jdk21.0.12+10-linux-amd64.tar.gz
sudo mv jdk-21.0.12 jdk-21
sudo mv jdk-21 /data/tools/jdk
# JDK 25
tar -xzf bellsoft-jdk25.0.4+9-linux-amd64.tar.gz
sudo mv jdk-25.0.4 jdk-25
sudo mv jdk-25 /data/tools/jdk

环境变量

bash
# 新建环境变量脚本
sudo touch /etc/profile.d/java.sh
# 打开编辑环境变量脚本
sudo vi /etc/profile.d/java.sh

增加以下内容

bash
export JAVA_11_HOME=/data/tools/jdk/jdk-11
export JAVA_21_HOME=/data/tools/jdk/jdk-21
export JAVA_25_HOME=/data/tools/jdk/jdk-25
export JAVA_HOME=$JAVA_25_HOME
export PATH=$JAVA_HOME/bin:$PATH
